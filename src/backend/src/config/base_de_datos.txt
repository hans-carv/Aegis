CREATE TABLE IF NOT EXISTS gatos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre VARCHAR(100) NOT NULL,
    edad_meses INT,
    color_pelaje VARCHAR(50),
    sexo VARCHAR(10) CHECK (sexo IN ('MACHO', 'HEMBRA')),
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS ingresos_felinos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    gato_id UUID NOT NULL REFERENCES gatos(id) ON DELETE CASCADE,
    via_procedencia VARCHAR(20) NOT NULL CHECK (via_procedencia IN ('RESCATADO', 'FUNDACION', 'PARTICULAR')),
    nombre_institucion VARCHAR(150),
    fecha_derivacion DATE,
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by UUID NOT NULL REFERENCES usuarios(id) ON DELETE RESTRICT,

    CONSTRAINT check_fundacion_fields CHECK (
        (via_procedencia = 'FUNDACION' AND nombre_institucion IS NOT NULL AND fecha_derivacion IS NOT NULL)
        OR (via_procedencia <> 'FUNDACION')
    )
);
