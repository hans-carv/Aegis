CREATE TABLE IF NOT EXISTS public.franjas_horarias (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    dia_semana VARCHAR(10) NOT NULL CHECK (dia_semana IN ('LUNES', 'MARTES', 'MIERCOLES', 'JUEVES', 'VIERNES', 'SABADO', 'DOMINGO')),
    hora_inicio TIME NOT NULL,
    hora_fin TIME NOT NULL,
    aforo_maximo INTEGER NOT NULL CHECK (aforo_maximo > 0),
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    motivo_bloqueo VARCHAR(255) NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_rango_horario CHECK (hora_inicio < hora_fin)
);

-- Índice para optimizar consultas de solapamiento
CREATE INDEX idx_franjas_dia_horario ON public.franjas_horarias (dia_semana, hora_inicio, hora_fin) WHERE activo = TRUE;

CREATE TABLE IF NOT EXISTS public.reservas_visita (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    usuario_id UUID NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    franja_id UUID NOT NULL REFERENCES public.franjas_horarias(id) ON DELETE RESTRICT,
    fecha_reserva DATE NOT NULL,
    cantidad_personas INTEGER NOT NULL DEFAULT 1 CHECK (cantidad_personas > 0),
    estado VARCHAR(20) NOT NULL DEFAULT 'CONFIRMADA' CHECK (estado IN ('CONFIRMADA', 'CANCELADA', 'COMPLETADA')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    -- Constraint única para satisfacer SCRUM-48 y Criterio 4:
    CONSTRAINT uq_usuario_franja_fecha UNIQUE (usuario_id, franja_id, fecha_reserva)
);

CREATE INDEX IF NOT EXISTS idx_reservas_franja_fecha 
ON public.reservas_visita (franja_id, fecha_reserva) 
WHERE estado = 'CONFIRMADA';

CREATE TABLE IF NOT EXISTS public.postulaciones_adopcion (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    usuario_id UUID NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    gato_id UUID NOT NULL REFERENCES public.gatos(id) ON DELETE RESTRICT,
    estado VARCHAR(20) NOT NULL DEFAULT 'EN_REVISION' CHECK (estado IN ('EN_REVISION', 'APROBADA', 'RECHAZADA', 'CANCELADA')),
    cuestionario_vivienda JSONB NOT NULL,
    presupuesto_mensual NUMERIC(10, 2) NOT NULL CHECK (presupuesto_mensual > 0),
    telefono_contacto VARCHAR(20) NOT NULL,
    direccion_domicilio VARCHAR(255) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_postulaciones_gato_id ON public.postulaciones_adopcion(gato_id);
CREATE INDEX IF NOT EXISTS idx_postulaciones_usuario_id ON public.postulaciones_adopcion(usuario_id);
CREATE INDEX IF NOT EXISTS idx_postulaciones_estado ON public.postulaciones_adopcion(estado);

-- Tabla para almacenar los metadatos criptográficos de los archivos
CREATE TABLE IF NOT EXISTS public.documentos_postulacion (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    postulacion_id UUID NOT NULL REFERENCES public.postulaciones_adopcion(id) ON DELETE CASCADE,
    tipo_documento VARCHAR(50) NOT NULL CHECK (tipo_documento IN ('CARNET_IDENTIDAD', 'FACTURA_SERVICIO', 'CROQUIS_DOMICILIO')),
    nombre_archivo_original VARCHAR(255) NOT NULL,
    mime_type VARCHAR(100) NOT NULL,
    ruta_cifrada TEXT NOT NULL,
    iv_vector VARCHAR(64) NOT NULL,
    hash_sha256 VARCHAR(64) NOT NULL,
    tamano_bytes BIGINT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_postulacion_tipo_documento UNIQUE (postulacion_id, tipo_documento)
);

CREATE INDEX IF NOT EXISTS idx_documentos_postulacion_id ON public.documentos_postulacion(postulacion_id);
