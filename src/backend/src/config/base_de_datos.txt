-- 4. Tabla: gatos
CREATE TABLE IF NOT EXISTS public.gatos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    codigo_felino VARCHAR(50) NOT NULL UNIQUE,
    nombre VARCHAR(100) NOT NULL,
    peso NUMERIC(5,2) NOT NULL DEFAULT 1.0 CHECK (peso > 0),
    talla VARCHAR(20),
    edad_meses INT NOT NULL DEFAULT 12 CHECK (edad_meses > 0),
    temperamento VARCHAR(150),
    color_pelaje VARCHAR(50),
    sexo VARCHAR(10) CHECK (sexo IN ('MACHO', 'HEMBRA')),
    estado VARCHAR(20) NOT NULL DEFAULT 'CUARENTENA' CHECK (estado IN ('CUARENTENA', 'DISPONIBLE', 'ADOPCION', 'ADOPTADO')),
    foto_url TEXT,
    historia TEXT,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_gatos_codigo_felino ON public.gatos (codigo_felino);

-- 5. Tabla: ingresos_felinos
CREATE TABLE IF NOT EXISTS public.ingresos_felinos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    gato_id UUID NOT NULL REFERENCES public.gatos(id) ON DELETE CASCADE,
    via_procedencia VARCHAR(20) NOT NULL CHECK (via_procedencia IN ('RESCATADO', 'FUNDACION', 'PARTICULAR')),
    nombre_institucion VARCHAR(150),
    fecha_derivacion DATE,
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by UUID NOT NULL REFERENCES public.usuarios(id) ON DELETE RESTRICT,
    CONSTRAINT check_fundacion_fields CHECK (
        (via_procedencia = 'FUNDACION' AND nombre_institucion IS NOT NULL AND fecha_derivacion IS NOT NULL)
        OR (via_procedencia <> 'FUNDACION')
    )
);

    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    postulacion_id UUID NOT NULL UNIQUE REFERENCES public.postulaciones_adopcion(id) ON DELETE RESTRICT,
    gato_id UUID NOT NULL REFERENCES public.gatos(id) ON DELETE RESTRICT,
    adoptante_id UUID NOT NULL REFERENCES public.usuarios(id) ON DELETE RESTRICT,
    administrador_id UUID NOT NULL REFERENCES public.usuarios(id) ON DELETE RESTRICT,
    direccion_tenencia TEXT NOT NULL,
    telefono_tenencia VARCHAR(30) NOT NULL,
    observaciones_entrega TEXT,
    estado_vinculo VARCHAR(20) NOT NULL DEFAULT 'VIGENTE' CHECK (estado_vinculo IN ('VIGENTE', 'DEVOLUCION_REGISTRADA')),
    fecha_adopcion TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 2. DoD 3: Constraint de unicidad activa en BD (un felino no puede tener dos enlaces VIGENTES simultáneos)
CREATE UNIQUE INDEX IF NOT EXISTS idx_unico_vinculo_adopcion_vigente
ON public.adopciones_formales (gato_id)
WHERE estado_vinculo = 'VIGENTE';

CREATE INDEX IF NOT EXISTS idx_adopciones_formales_adoptante ON public.adopciones_formales(adoptante_id);

-- 3. Inmutabilidad histórica: impedir alteración de gato_id, adoptante_id, postulacion_id o fecha_adopcion
CREATE OR REPLACE FUNCTION public.proteger_inmutabilidad_adopcion()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.gato_id <> OLD.gato_id
       OR NEW.adoptante_id <> OLD.adoptante_id
       OR NEW.postulacion_id <> OLD.postulacion_id
       OR NEW.fecha_adopcion <> OLD.fecha_adopcion THEN
        RAISE EXCEPTION 'El vínculo histórico de adopción es inmutable y no permite modificar sus claves o fecha original.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_inmutabilidad_adopcion ON public.adopciones_formales;
CREATE TRIGGER trg_inmutabilidad_adopcion
BEFORE UPDATE ON public.adopciones_formales
FOR EACH ROW
EXECUTE FUNCTION public.proteger_inmutabilidad_adopcion();
